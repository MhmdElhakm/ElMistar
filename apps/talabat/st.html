ربط جوجل شيتخطوات سهلة

سحب فقط (بدون برمجة)

1- افتح الشيت ← مشاركة ← "أي شخص لديه الرابط يمكنه العرض".
2- الصق رابط الشيت في الإعدادات ثم "سحب من الشيت" (تخطّي المكرر بنفس الاسم والهاتف).
الأعمدة المقبولة: الاسم، الهاتف، الصف، المجموعة، اليوم، الموعد، المستوى، المدفوع، ملاحظات.

مزامنة كاملة (رفع + سحب)

1- من الشيت: الإضافات ← Apps Script ← امسح كل شيء والصق الكود التالي ← نشر ← "نشر كتطبيق ويب" مع: التنفيذ باسمك (Me) + الوصول "أي شخص" (Anyone) — بدونهما يطلب الرابط تسجيل دخول ويفشل.
2- الصق رابط الـ Web App (المنتهي بـ /exec) في الإعدادات ثم "اختبار الاتصال" ثم "مزامنة الآن".
⚠️ بعد أي تعديل في الكود: نشر ← إدارة عمليات النشر ← تعديل ← اختر "إصدار جديد" ← نشر (وإلا يظل الكود القديم يعمل).

var SHEET_STUDENTS='students',SHEET_GROUPS='groups';
function doGet(e){
  var a=(e&&e.parameter&&e.parameter.action)||'pull';
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  if(a==='pull')return json({ok:true,students:readAll(ss,SHEET_STUDENTS),groups:readAll(ss,SHEET_GROUPS)});
  return json({ok:false,error:'unknown action'});
}
function doPost(e){
  var d={};try{d=JSON.parse(e.postData.contents);}catch(err){}
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  if(d.action==='push'){
    if(d.students)writeAll(ss,SHEET_STUDENTS,d.students);
    if(d.groups)writeAll(ss,SHEET_GROUPS,d.groups);
    return json({ok:true,saved:(d.students||[]).length});
  }
  return json({ok:false});
}
function json(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);}
function readAll(ss,name){
  var sh=ss.getSheetByName(name);if(!sh)return [];
  var v=sh.getDataRange().getValues();if(v.length<2)return [];
  var h=v[0].map(String),out=[];
  for(var i=1;i<v.length;i++){var o={};for(var j=0;j<h.length;j++)o[h[j]]=v[i][j];out.push(o);}
  return out;
}
function writeAll(ss,name,arr){
  var sh=ss.getSheetByName(name)||ss.insertSheet(name);
  sh.clear();
  if(!arr.length){sh.appendRow(['empty']);return;}
  var keys=Object.keys(arr[0]);sh.appendRow(keys);
  arr.forEach(function(o){sh.appendRow(keys.map(function(k){var x=o[k];return (x&&typeof x==='object')?JSON.stringify(x):x;}));});
  try{
    var lastRow=sh.getLastRow(),lastCol=sh.getLastColumn();
    sh.getRange(1,1,1,lastCol).setFontWeight('bold').setBackground('#1e293b').setFontColor('#ffffff');
    sh.setFrozenRows(1);
    var gi=keys.indexOf('الصف');if(gi<0)gi=keys.indexOf('grade');
    if(gi>=0&&lastRow>1){
      var pal={'مرحلة الكي جي والتأسيس':'#FEF3C7','الصف الأول الابتدائي':'#DBEAFE','الصف الثاني الابتدائي':'#E9D5FF','الصف الثالث الابتدائي':'#CCFBF1','الصف الرابع الابتدائي':'#FFE4E6','الصف الخامس الابتدائي':'#DCFCE7','الصف السادس الابتدائي':'#FFEDD5'};
      for(var r=2;r<=lastRow;r++){var gv=String(sh.getRange(r,gi+1).getValue()||'').trim();var c=pal[gv]||'#F1F5F9';sh.getRange(r,1,1,lastCol).setBackground(c);}
    }
    for(var cc=1;cc<=lastCol;cc++){try{sh.autoResizeColumn(cc);}catch(e){}}
  }catch(e){}
}
